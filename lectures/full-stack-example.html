// lib/db.ts — Database query

export function getCoursesWithEnrollment() {
  const stmt = db.prepare(`
    SELECT
      courses.*,
      COUNT(registrations.id) as enrolled_count
    FROM courses
    LEFT JOIN registrations ON courses.id = registrations.course_id
    GROUP BY courses.id
  `);
  return stmt.all();
}

// app/CoursesPage.tsx

async function handleRegister(courseId: number) {
  // Send POST request to our API
  const response = await fetch('/api/courses/register', {
    method: 'POST',                              // POST = "create something"
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ courseId })           // Send course ID as JSON
  });

  // Check if it worked
  if (response.ok) {
    fetchCourses();  // Refresh the course list to show updated count
  } else {
    const data = await response.json();
    setError(data.error);  // Show error like "Course is full"
  }
}

Step 3: API Route Receives Request

Next.js routes POST to app/api/courses/register/route.ts

// app/api/courses/register/route.ts

import { NextRequest, NextResponse } from 'next/server';
import { registerForCourse } from '@/lib/db';
import { getSession } from '@/lib/auth';

export async function POST(request: NextRequest) {
  // 1. Check if user is logged in (read session from cookie)
  const session = await getSession();
  if (!session) {
    return NextResponse.json({ error: 'Not logged in' }, { status: 401 });
  }

  // 2. Get courseId from request body
  const { courseId } = await request.json();

  // 3. Try to register (this talks to the database)
  const result = registerForCourse(session.id, courseId);

  // 4. Return success or error
  if (!result.success) {
    return NextResponse.json({ error: result.error }, { status: 400 });
  }

  return NextResponse.json({ success: true });
}

Step 4: Database Transaction (The Critical Part!)

We must check seats AND insert in one atomic operation

// lib/db.ts

export function registerForCourse(userId: number, courseId: number) {
  // Transaction = all steps succeed or all fail together
  const checkAndRegister = db.transaction(() => {

    // Step A: How many students are enrolled?
    const { count } = db.prepare(
      'SELECT COUNT(*) as count FROM registrations WHERE course_id = ?'
    ).get(courseId);

    // Step B: What's the max?
    const course = db.prepare(
      'SELECT max_students FROM courses WHERE id = ?'
    ).get(courseId);

    // Step C: Is there room?
    if (count >= course.max_students) {
      return { success: false, error: 'Course is full' };
    }

    // Step D: Register the student!
    db.prepare(
      'INSERT INTO registrations (user_id, course_id) VALUES (?, ?)'
    ).run(userId, courseId);

    return { success: true };
  });

  return checkAndRegister();  // Execute the transaction
}

🚨 Why Do We Need a Transaction?

❌ Without Transaction (Bug!)

Time 0: Course has 1/2 seats

Time 1: Alice checks: "1 enrolled, room for 1 more" ✓

Time 2: Bob checks: "1 enrolled, room for 1 more" ✓

Time 3: Alice inserts her registration

Time 4: Bob inserts his registration

Result: 3 students in 2-seat course! 💥

✅ With Transaction (Safe!)

Time 0: Course has 1/2 seats

Time 1: Alice starts transaction (locks rows)

Time 2: Alice checks + inserts atomically

Time 3: Alice commits. Now 2/2 enrolled.

Time 4: Bob starts transaction, checks: "2/2 = FULL"

Result: Bob gets "Course is full" error ✓

db.transaction() makes check + insert atomic — no one can sneak in between!

Server vs Client Components

// app/page.tsx
// No 'use client' = runs on server

export default async function Page() {
  // ✅ Can access database
  const data = await db.query(...);

  // ❌ Cannot use useState

'use client';
// Runs in browser

export default function Button() {

How Login Works (Simplified)

1
User submits login form POST /api/auth/login { username: "alice", password: "1234" }
2
Server checks password bcrypt.compare(password, user.password_hash) → true/false
3
Server creates JWT token JWT contains: { id: 2, username: "alice", is_admin: false }
4
Server sets cookie Set-Cookie: session=eyJhbGci...; HttpOnly; Secure
5
Future requests: browser sends cookie automatically getSession() reads cookie → verifies JWT → returns user info
